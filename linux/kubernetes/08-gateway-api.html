安装 nginx-gateway-fabric

  • 使用 Helm 安装 NGINX Gateway Fabric
# 
mkdir ngfdir
cd ngfdir/

# 下载 2.6.7 版本文件。拉取 OCI 格式 Helm chart，解压到本地
helm pull oci://ghcr.io/nginx/charts/nginx-gateway-fabric --version 2.8.1 --untar
cd nginx-gateway-fabric/

helm install ngf . \
  --create-namespace -n nginx-gateway \
  --set nginx.service.type=LoadBalancer \
  --set nginx.service.externalTrafficPolicy=Cluster \
  --set nginxGateway.snippetsFilters.enable=true

#  --set nginx.service.type=LoadBalancer 使用负载均衡器。会自动分配公网 LB
#  --set nginx.service.externalTrafficPolicy=Cluster 外部流量转发策略。
#    Cluster（这里配置的）：源 IP 会被 SNAT 改写为节点 IP；可以做负载均衡跨节点转发。
#    Local：保留真实客户端源 IP，但流量只能落到后端 Pod 所在节点。
#  --set nginxGateway.snippets.enable=true  NGF 的扩展能力，可以在 Gateway API 资源中嵌入原生 Nginx 配置片段（nginx.conf 片段）。同时启用 SnippetsFilter 和 SnippetsPolicy。

#或者将配置写到 value 文件中

cat <<EOF> values-dev.yaml
#  NGF 的扩展能力
nginxGateway:
  snippets:
    enable: true
# nginxproxy 配置
nginx:
  replicas: 1 # 默认 1 个
  service:
    type: LoadBalancer
    externalTrafficPolicy: Cluster
EOF

时区与日志格式化

#nginx-gateway-fabric 2.6.7

cat >values-ngf.yaml <<\EOF
nginxGateway:
  snippets:
    enable: true

nginx:
  service:
    type: LoadBalancer
    externalTrafficPolicy: Cluster
  patches:
  - type: JSONPatch
    value:
    - op: add
      path: /spec/template/spec/volumes/-
      value:
        name: tz-shanghai
        hostPath:
          path: /usr/share/zoneinfo/Asia/Shanghai
          type: File
    - op: add
      path: /spec/template/spec/containers/0/volumeMounts/-
      value:
        name: tz-shanghai
        mountPath: /etc/localtime
        readOnly: true

  config:
    logging:
      accessLog:
        escape: json
        format: >-
          {"time":"$time_iso8601","remote_addr":"$remote_addr","x_forwarded_for":"$http_x_forwarded_for","host":"$host","method":"$request_method","uri":"$request_uri","protocol":"$server_protocol","status":$status,"body_bytes_sent":$body_bytes_sent,"request_length":$request_length,"request_time":$request_time,"upstream_addr":"$upstream_addr","upstream_status":"$upstream_status","upstream_response_time":"$upstream_response_time","referer":"$http_referer","user_agent":"$http_user_agent","request_id":"$request_id"}

配置说明 

# 配置说明
#nginxGateway.snippets.enable 同时启用 SnippetsFilter 和 SnippetsPolicy。
#nginxGateway.snippetsFilters.enable 已被 chart 标记为 deprecated，且只开前者
#nginx.service.type=LoadBalancer 使用负载均衡器
#nginx.service.externalTrafficPolicy=Cluster 外部流量转发策略
#  Cluster（这里配置的）：源 IP 会被 SNAT 改写为节点 IP；可以做负载均衡跨节点转发。
#  Local：保留真实客户端源 IP，但流量只能落到后端 Pod 所在节点。

# ---- 时区：东八区 ----
# 数据面镜像是 Alpine，既没有 /etc/localtime 也没有 tzdata，
# 所以 $time_iso8601 / $time_local / error log 的时间全是 UTC
# （日志里是 +00:00，比北京时间晚 8 小时）。
#
# 只设 TZ=Asia/Shanghai 是没用的：musl 会去找
# /usr/share/zoneinfo/Asia/Shanghai，镜像里没有这个文件，就静默回退 UTC。
# 必须把宿主机的时区文件挂进来。这里刻意不设 TZ 环境变量——
# musl 在 TZ 未设时直接读 /etc/localtime，正好是我们挂进去的那个。
#
# 用 hostPath 挂宿主机的 /usr/share/zoneinfo/Asia/Shanghai 这个实际文件
#
# NginxProxy CRD 的 container 字段只有 debug/hostPorts/image，给不了
# env 和 volumeMounts，所以只能走 patches。
# chart 会把 nginx.patches 渲染进 NginxProxy 的 kubernetes.deployment.patches。
# 必须用 JSONPatch，不能用 StrategicMerge（默认类型）。
# StrategicMerge 在 NGF provisioner 这条路径上会**按索引覆盖**而不是按 name 合并：
# 只含一个元素的 volumes 会盖掉 volumes[0]（原本是 projected 类型的 token），
# API Server 报 "may not specify more than 1 volume type"，
# 同时 volumeMounts 的引用全部错位，Deployment 更新直接被拒。
# JSONPatch 的 "/-" 是追加到列表末尾，不碰任何现有元素。
#
# containers/0 就是 nginx：数据面 Pod 只有这一个业务容器（另有 init 初始化容器）。

# ---- 日志 json 格式化 ----
# nginx.config 的内容会被原样渲染进 NginxProxy 的 spec
# （chart 模板 templates/nginxproxy.yaml: toYaml .Values.nginx.config）
# nginx.config.logging.accessLog.escape=json —— 让 nginx 按 JSON 规则转义变量值（引号、反斜杠、控制字符），
# 否则 User-Agent 里带引号就会把整行 JSON 破坏掉。
# 注意两条硬性限制（来自 NginxProxy CRD 的 schema）：
#   1. 不能有单引号 —— 这段会被渲染进单引号包裹的 log_format 指令里
#   2. 不能有换行 —— 必须写成一行
#
# 字段取舍：$status / $body_bytes_sent / $request_length / $request_time
# 这几个 nginx 保证非空，所以不加引号，让它们在 JSON 里是数字类型，
# Grafana 里可直接做数值比较和聚合。
# upstream 相关变量在未走到上游时会是空值，必须加引号，否则
# 会产生 "upstream_status":, 这种非法 JSON。

# 查看有没有生效
kubectl -n nginx-gateway get nginxproxies.gateway.nginx.org  ngf-proxy-config -oyaml

# 自定义 nginxproxy
cat <<\EOF> proxy-public-config.yaml

kind: NginxProxy
metadata:

    app.kubernetes.io/version: 2.8.1

  namespace: gw

  # 信任 NLB 发来的 Proxy-Protocol v2
  # in-tree NLB(instance 目标)+ externalTrafficPolicy: Cluster,
  # kube-proxy 会 SNAT,pod 看到的源是节点 IP
  rewriteClientIP:
    mode: ProxyProtocol
    trustedAddresses:
    - type: CIDR
      value: 172.31.0.0/16
  logging:
    accessLog:
      escape: json
      format: '{"time":"$time_iso8601", "cf_ray":"$http_cf_ray", "request_id":"$request_id",
        "cf_connecting_ip":"$http_cf_connecting_ip","remote_addr":"$remote_addr",
        "x_forwarded_for":"$http_x_forwarded_for", "cf_country":"$http_cf_ipcountry",
        "method":"$request_method", "scheme":"$scheme","host":"$host", "path":"$uri",
        "protocol":"$server_protocol", "status":$status, "request_length":$request_length,
        "body_bytes_sent":$body_bytes_sent,"request_time":$request_time, "upstream_addr":"$upstream_addr",
        "upstream_connect_time":"$upstream_connect_time","upstream_header_time":"$upstream_header_time",
        "upstream_response_time":"$upstream_response_time", "upstream_status":"$upstream_status","referer":"$http_referer",
        "user_agent":"$http_user_agent", "xff_prefix":"$http_x_forwarded_prefix",
        "request_uri":"$request_uri"}'
    agentLevel: error
    errorLevel: warn
  kubernetes:
    deployment:
      container:
        image:
          pullPolicy: IfNotPresent
          repository: ghcr.io/nginx/nginx-gateway-fabric/nginx

NGF 数据面的 gzip 压缩

数据面的 nginx.conf 由 NGF 控制器生成，不能直接改。注入配置有两条路：

  • SnippetsFilter —— 作用于 HTTPRoute/GRPCRoute，要在每条路由里引用
  • SnippetsPolicy —— 作用于 Gateway，一次配置对该 Gateway 下所有路由生效 <- 用这个

两者都由 helm 的 nginxGateway.snippets.enable 开启（旧的 snippetsFilters.enable 已废弃，且只开 SnippetsFilter）。

gzip 压缩说明 

gzip on;

# 4 是性价比拐点：再往上 CPU 涨得快、体积只多省几个百分点
gzip_comp_level 4;

# 小响应压了反而更大（gzip 头部开销约 20 字节）
gzip_min_length 1k;

# 必须显式开。默认 off 时，带 Via 头的代理响应一律不压——
# 而网关本身就是反向代理，不加这行等于没开
gzip_proxied any;

# 让 CDN / 中间代理按 Accept-Encoding 分别缓存，
# 避免把压缩版发给不支持的客户端
gzip_vary on;

# text/html 不用列，nginx 总是压缩它。
# 刻意【不包含】已经是压缩格式的类型：
#   图片 png/jpg/gif/webp、字体 woff2、视频、zip/gz
# 对它们再压一遍只烧 CPU，体积还可能变大。
gzip_types
  text/plain
  text/css
  text/xml
  text/javascript
  application/json
  application/javascript
  application/xml
  application/rss+xml
  application/atom+xml
  application/manifest+json
  application/vnd.api+json
  image/svg+xml;

AWS 公网 NLB

下面配置 externalTrafficPolicy: Cluster 是 Cluster，想要获得客户端真实 IP 需要 AWS 上负载均衡器的目标组开启 Proxy‑Protocol 。如果是 Local 建议开启跨区域负载均衡，因为空着的 AZ 会成为流量黑洞，所以必须靠跨区兜底。

  • Cluster 如果开了跨区负载会多付跨 AZ 流量费却没好处：Cluster 模式下 kube-proxy 本来就在全集群的 endpoint 里挑 pod，跨 AZ 转发已经发生了；NLB 这层再做一次跨区只是多一跳。

如果生成 NBL 负载均衡分两种情况：

  • 安装了 AWS LBC：使用 aws-load-balancer-type: "external" ， 目标组为 pod ip，同时生成专属安全组。需要开启跨区域负载均衡， NLB 注册的是 pod IP，不是节点。所以健康目标的分布取决于 pod 落在哪些 AZ，而不是节点分布。
  • 未安装 AWS LBC：使用 aws-load-balancer-type: nlb ， 目标组为 节点 IP，使用节点已有的安全组。不需要开启跨区域负载均衡
  • 安装了 AWS LBC
    • service 使用 externalTrafficPolicy: Cluster
    • service 注解 aws-load-balancer-type: "external"
    • 需要开启跨区域负载均衡。
    • 自动创建安全组。

apiVersion: gateway.nginx.org/v1alpha1
kind: SnippetsFilter
metadata:
  name: global-security
  namespace: default
spec:
  snippets:
  # http.server 上下文 == ingress-nginx 的 server-snippet，语义一一对应。
  # $request_uri 是未经解码的原始 URI（含 query string），~* 为不区分大小写的
  # 子串匹配 —— /api/actuator/health、/ACTUATOR 都会被拦，这点和原来一致，
  # 用 PathPrefix 的 HTTPRoute 做不到等价。
  - context: http.server
    value: |
      if ($request_uri ~* "/actuator")  { return 403; }
      if ($request_uri ~* "snail-?job") { return 403; }
      if ($http_x_forwarded_prefix ~* "actuator|snail-?job") { return 403; }
      if ($request_uri ~*  "~*/(v[23]/api-docs|swagger-ui|swagger-resources|knife4j|webjars)(/|\?|$)") {return 403; }

      # 路径穿越 / 多重 URL 编码绕过
      #
      # 捕获：GET /api/basic/..%252fchain/v2/zz9608t  → 200
      #   %25 = '%'，nginx 解码一次得到 ..%2f，对它而言只是普通字符串，
      #   内建的路径规范化不会触发，请求被原样转给后端；
      #   后端（尤其 Spring Cloud Gateway）若再解码一次就变成 ../，
      #   可用于在微服务之间横向跳转。
      #
      # 用 $request_uri（未解码的原始 URI）匹配，才能看到 %252f 这层。
      # 只拦 ".." 紧跟分隔符的形式，不拦裸 %25 / %2f ——
      # 那两个在搜索参数、含斜杠的资源 ID 里是合法的，会误伤。
      if ($request_uri ~* "\.\.(/|%2f|%252f|%5c|%255c)") { return 403; }
      if ($request_uri ~* "(%2e%2e|%252e)")              { return 403; }

      # ── 支付状态 / 金额篡改参数 ──────────────────────────────────
      # 捕获： 拿真实收银台会话，在 03:54 一分钟内连打
      #   ?payStatus=1  ?payStatus=PAID  ?tradeStatus=SUCCESS  ?success=true
      #   ?amount=999   ?amount=0.01     ?payAmount=1          ?sign=1
      # 04:10 又补了方法覆盖 ?_method=POST&status=1&tradeStatus=SUCCESS
      # 和参数污染 ?status=1&status=0、数组写法 ?status[]=1。
      if ($args ~* "(^|&)_method=")                                             { return 403; }
      if ($args ~* "(^|&)(pay_?status|trade_?status|order_?status|payment_?status)=")  { return 403; }
      if ($args ~* "(^|&)(pay_?amount|total_?amount|real_?amount|paid_?amount)=")      { return 403; }
      if ($args ~* "(^|&)(mock|sandbox|test_?mode|debug)=(1|true|yes|on)")      { return 403; }

      # ── 扫描工具 UA ──────────────────────────────────────────────
      # 只封明确自报家门的攻击工具。**不按爬虫 UA 封** —— 34.48 用了 24 个
      # 伪造爬虫 UA（Googlebot/GPTBot/Claude-User/Kimi/Qwenbot…），按 UA 封拦不住
      # 他（换一个就行），却会误伤真实 Googlebot 影响 SEO。
      # 伪造爬虫的正确解法是 Cloudflare 的 Verified Bots（按 ASN + 反向解析校验）。
      if ($http_user_agent ~* "(sqlmap|nikto|nuclei|acunetix|nessus|openvas|wpscan|dirbuster|gobuster|feroxbuster|ffuf|masscan|zgrab|nmap|hydra|metasploit|xray)") { return 403; }

      # ── 从不合法的 HTTP 方法 ─────────────────────────────────────
      # TRACE/TRACK 可用于 XST 跨站追踪，CONNECT 是代理方法，正常业务永远用不到。
      # 注意这里没有拦 PUT/PATCH/DELETE —— 那几个在其他 REST 接口上是合法的，
      # 全局拦会误伤；"只读支付查询接口上出现写方法"由 Loki 告警检测。
      if ($request_method ~ "^(TRACE|TRACK|CONNECT)$") { return 405; }

      # ── 扫描器 payload 字典（34的 352 个请求）───────────
      # 这些路径在平台上完全不存在，出现即为探测。
      # 上面的 ".." 规则已经挡下了当晚的全部样本，这里补的是不带 ../ 的变体。
      if ($request_uri ~* "/proc/self/(environ|cmdline|status|fd)")  { return 403; }
      if ($request_uri ~* "/userfiles(/|\?|$)")                      { return 403; }
      if ($request_uri ~* "jobs_u/get_log_file")                     { return 403; }
      if ($request_uri ~* "/api/console/api_server")                 { return 403; }
      if ($request_uri ~* "(wp-login|wp-admin|wp-content|xmlrpc\.php)")  { return 403; }
      if ($request_uri ~* "/(_profiler|telescope|phpinfo|server-status|solr|phpmyadmin|adminer)(/|\?|$)")  { return 403; }
      if ($request_uri ~* "[?&](rest_route|input_file|SPX_KEY|SPX_UI_URI|cmd|exec)=")  { return 403; }

      # ── 按真实访客 IP 封禁 ─────────────────
      if ($http_cf_connecting_ip = "x.x.x.x")     { return 403; }  # GCP：12 分钟 352 请求，24 个伪造爬虫 UA
      if ($http_cf_connecting_ip = "x.x.x.x1")    { return 403; }  # Cogent：支付篡改 + 目录爆破
      if ($http_cf_connecting_ip = "x.x.x.x2") { return 403; }  # Oracle(IN)：actuator 注入 + 凭据爆破

  # X-Forwarded-Prefix 注入防护（原 ingress-nginx 没有，是迁移中发现的真实攻击）
  #
  # 捕获到的探测：
  #   X-Forwarded-Prefix: /socket/actuator/env
  #   GET /api?Authorization=...        cf_country=MY  status=200
  #
  # Spring Cloud Gateway 用这个头重建请求路径，攻击者借此绕过基于
  # $request_uri 的过滤直达内部 actuator。原规则只看 URI，完全没命中。
  #
  # 两层防护缺一不可：
  #   · 下面的 proxy_set_header 把头清空 —— 后端收不到，堵死利用链
  #   · 上面的 if 当场返回 403      —— 在日志里留痕，才能告警
  # 只有前者的话攻击确实被挡住了，但日志里是 200，看不出有人在打。
  - context: http.server.location
    value: |
      proxy_set_header X-Forwarded-Prefix "";

---
# 通用的"拒绝"过滤器 —— 给兜底规则用，让未显式放行的路径返回 403 而不是 404。
#
# 用法：在 HTTPRoute 里加一条 PathPrefix: / 的规则引用它。
# Gateway API 按最长前缀匹配，具体路径的规则会压过 /，所以白名单照常工作。
#
# ⚠ 兜底规则必须保留 backendRefs（随便指一个存在的 Service）。
#   NGF 对没有 backendRef 的规则会渲染成 return 500，和这里的 return 403
#   撞车且先后顺序不可控。留占位 backend 后，return 403 在 rewrite 阶段
#   就短路了，proxy_pass 永远不会执行，不会真的产生回源流量。
apiVersion: gateway.nginx.org/v1alpha1
kind: SnippetsFilter
metadata:
  name: deny-all
  namespace: default
spec:
  snippets:
  - context: http.server.location
    value: |
      return 403;
EOF

cat <<\EOF> httproute-openapi.yaml
---
apiVersion: gateway.networking.k8s.io/v1
kind: HTTPRoute
metadata:
  name: in-openapi
  namespace: default
spec:
  hostnames:
  - openapi.test.jp
  parentRefs:
  - name: gateway-internal
    namespace: gw
    sectionName: http
  rules:
  - backendRefs:
    # nginx 内部配置做安装兜底
    - name: jp-gateway-svc
      port: 8080
    matches:
    - path:
        type: PathPrefix
        value: /api/pay
    - path:
        type: PathPrefix
        value: /api/pay/createOrder
    - path:
        type: PathPrefix
        value: /api/pay/payout
    filters:
    - type: ExtensionRef
      extensionRef: {group: gateway.nginx.org, kind: SnippetsFilter, name: global-security}

  # ── 兜底：其余路径 403
  - matches:
    - path: {type: PathPrefix, value: /}
    filters:
    - type: ExtensionRef
      extensionRef: {group: gateway.nginx.org, kind: SnippetsFilter, name: deny-all}
    backendRefs:
    - name: jp-gateway-svc
      port: 8080
EOF

apiVersion: gateway.networking.k8s.io/v1
kind: HTTPRoute
metadata:
  name: in-admin
  namespace: default
spec:
  hostnames:
  - admin.test.jp
  parentRefs:
  - name: gateway-internal
    namespace: gw
    sectionName: http
  rules:
  - backendRefs:
    - name: admin-svc
      port: 80
    matches:
    - path:
        type: PathPrefix
        value: /
    filters:
    - type: ExtensionRef
      extensionRef:
        group: gateway.nginx.org
        kind: SnippetsFilter
        name: global-security
  - backendRefs:
    - name: admin-server-svc
      port: 8080
    matches:
    - path:
        type: PathPrefix
        value: /alpha/v1/
    filters:
    - type: ExtensionRef
      extensionRef: {group: gateway.nginx.org, kind: SnippetsFilter, name: global-security}
    - type: URLRewrite
      urlRewrite:
        path:
          type: ReplacePrefixMatch
          replacePrefixMatch: /
    - type: ExtensionRef
      extensionRef: {group: gateway.nginx.org, kind: SnippetsFilter, name: admin-sinppetsfilter}

---
apiVersion: gateway.nginx.org/v1alpha1
kind: ClientSettingsPolicy
metadata:
  name: admin-client-policy
  namespace: default
spec:
  targetRef:
    group: gateway.networking.k8s.io
    kind: HTTPRoute
    name: in-admin
  body:
    maxSize: "100m"
---
apiVersion: gateway.nginx.org/v1alpha1
kind: SnippetsFilter
metadata:
  name: admin-sinppetsfilter
  namespace: default
spec:
  snippets:
  - context: http.server.location
    value: |
      proxy_read_timeout 5m;
